import React, { useEffect, useRef } from "react";
import styled, { keyframes } from "styled-components";
import { motion } from "framer-motion";

const sponsors = [
  { logo: "/sponsors/Bosch_symbol_logo_black_red 3.png", name: "Bosch" },
  { logo: "/sponsors/Silent Logo White 1.png", name: "Silent Owl" },
  { logo: "/sponsors/Logo 1.png", name: "Contentstack" },
  { logo: "/sponsors/Group 248.png", name: "Pathway" },
  { logo: "/sponsors/Group 249.png", name: "UTS" },
  { logo: "/sponsors/Group 250.png", name: "EaseMyTrip" },
  { logo: "/sponsors/Group 253 (1).png", name: "Big TV" },
  { logo: "/sponsors/Group 253.png", name: "Institution's Innovation Council" },
  { logo: "/sponsors/Group 308.png", name: "Unstop" },
  { logo: "/sponsors/Group 317.png", name: "NMDC" },
  { logo: "/sponsors/image 64 (1).png", name: "Servo" },
  { logo: "/sponsors/image 64 (2).png", name: "QuillBot" },
];

// Keyframes for infinite scrolling
const scroll = keyframes`
  0% { transform: translateX(0); }
  100% { transform: translateX(-50%); }
`;

const SponsorsSection = () => {
  const carouselRef = useRef(null);

  // Duplicate sponsors array for seamless infinite scrolling
  const extendedSponsors = [...sponsors, ...sponsors];

  // Pause animation on hover
  const handleMouseEnter = () => {
    if (carouselRef.current) {
      carouselRef.current.style.animationPlayState = "paused";
    }
  };

  const handleMouseLeave = () => {
    if (carouselRef.current) {
      carouselRef.current.style.animationPlayState = "running";
    }
  };

  return (
    <SponsorsWrapper>
      <motion.div
        initial={{ opacity: 0, y: "50%" }}
        animate={{ opacity: 1, y: 0 }}
        transition={{ duration: 0.8 }}
      >
        <Title>Our Sponsors</Title>
        <Subtitle>Partnered with Tinkerers’ Lab IITH</Subtitle>
      </motion.div>

      <CarouselWrapper
        onMouseEnter={handleMouseEnter}
        onMouseLeave={handleMouseLeave}
      >
        <Carousel ref={carouselRef}>
          {extendedSponsors.map((sponsor, index) => (
            <SponsorCard key={index}>
              <SponsorLogo
                src={sponsor.logo}
                alt={`${sponsor.name} logo`}
                whileHover={{ scale: 1.1, boxShadow: "0 0 20px rgba(255, 255, 255, 0.5)" }}
                transition={{ type: "spring", stiffness: 300 }}
              />
              <SponsorName>{sponsor.name}</SponsorName>
            </SponsorCard>
          ))}
        </Carousel>
      </CarouselWrapper>
    </SponsorsWrapper>
  );
};

export default SponsorsSection;

const SponsorsWrapper = styled.section`
  padding: 6rem 0;
  background: linear-gradient(135deg, #1a1a2e 0%, #16213e 100%);
  position: relative;
  overflow: hidden;

  /* Subtle particle effect in background */
  &::before {
    content: "";
    position: absolute;
    top: 0;
    left: 0;
    width: 100%;
    height: 100%;
    background: radial-gradient(circle, rgba(255, 255, 255, 0.1) 1px, transparent 1px);
    background-size: 20px 20px;
    opacity: 0.3;
    animation: float 15s infinite linear;
  }

  @keyframes float {
    0% {
      transform: translateY(0);
    }
    50% {
      transform: translateY(-20px);
    }
    100% {
      transform: translateY(0);
    }
  }
`;

  text-align: center;

  font-family: "Inter", sans-serif;
  font-weight: 400;
  font-size: 1.5rem;

  width: 100%;
  overflow: hidden;
`;

const Carousel = styled.div`

  width: fit-content;
  animation: ${scroll} 30s linear infinite;
